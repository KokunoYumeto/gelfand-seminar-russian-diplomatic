\Unclear{GELFAND-P000981}{root-definition}{$g_\alpha$ наз. \underline{корнем}, если $[g_\alpha,h]=\alpha(h)g_\alpha$, $g_\alpha$}

\SourcePage{GELFAND-P000982}{12}

Тогда получается \underline{$(\alpha,\alpha)>0$.}

\end{equation*}
